async function extractPdfText(file){

  const buf = await file.arrayBuffer();

  const pdf = await pdfjsLib.getDocument({data: buf}).promise;

  const pages = [];

  for(let p=1;p<=pdf.numPages;p++){

    const page = await pdf.getPage(p);

    const content = await page.getTextContent();

    const rows = [];

    for(const item of content.items){

      const y = item.transform[5];

      let row = rows.find(r=>Math.abs(r.y-y) <= 2);

      if(!row){ row = {y, items:[]}; rows.push(row); }

      row.items.push({x:item.transform[4], text:item.str});

    }

    rows.sort((a,b)=>b.y-a.y).forEach(r=>r.items.sort((a,b)=>a.x-b.x));

    pages.push({

      height: page.getViewport({scale:1}).height,

      rows: rows.map(r=>({y:r.y, text:r.items.map(i=>i.text).join("").trim()})).filter(r=>r.text)

    });

  }


  const repeatedMargins = new Map();

  pages.forEach((page,pageIndex)=>{

    page.rows.forEach(row=>{

      const inMargin = row.y > page.height*.84 || row.y < page.height*.16;

      if(!inMargin) return;

      const key = row.text.replace(/\s+/g," ").trim().toLowerCase();

      if(!key) return;

      if(!repeatedMargins.has(key)) repeatedMargins.set(key, new Set());

      repeatedMargins.get(key).add(pageIndex);

    });

  });

  const noise = new Set([...repeatedMargins].filter(([,seen])=>seen.size>=2).map(([key])=>key));

  return pages.flatMap(page=>page.rows

    .filter(row=>!noise.has(row.text.replace(/\s+/g," ").trim().toLowerCase()))

    .map(row=>row.text)

  ).join("\n");

}

function parseQuestions(text){

  // Fix PDF merging issue: Force newline if a number is stuck at the end of a choice line

  text = text.replace(/([^\s])\s+(\d{1,3}\.\s+[A-Za-zก-๙])/g, "$1\n$2");

  

  const lines = text.replace(/\r\n?/g, "\n").split("\n");

  const blocks = [];

  let current = null;

  const qStart = /^\s*(\d{1,3})\s*[\.\)]\s*(.*\S)\s*$/;

  const isMetadata = value => /^(?:ข้อสอบ|นักศึกษา|หมายเหตุ)|(?:คะแนนเต็ม|จำนวนผู้สอบ|ฝ่ายวิชาการ|\bMAX\b|\bMIN\b|\bMEAN\b|\bS\.D\.)/i.test(value.trim());

  

  for(const raw of lines){

    const m = raw.match(qStart);

    if(m && parseInt(m[1],10) < 300 && !isMetadata(m[2])){

      if(current) blocks.push(current);

      current = {number: m[1], lines:[m[2]]};

    } else if(current){

      current.lines.push(raw);

    }

  }

  if(current) blocks.push(current);


  const choiceRe = /^\s*([A-E])[\.\)]\s*(.*)$/i;

  const questions = [];

  for(const b of blocks){

    const stemLines = [];

    const choices = {A:"",B:"",C:"",D:"",E:""};

    let inChoices = false;

    for(const ln of b.lines){

      const cm = ln.match(choiceRe);

      if(cm){

        inChoices = true;

        const letter = cm[1].toUpperCase();

        choices[letter] = (choices[letter] ? choices[letter]+" " : "") + cm[2].trim();

      } else if(inChoices){

        const letters = ["A","B","C","D","E"];

        for(let i=letters.length-1;i>=0;i--){

          if(choices[letters[i]]){ choices[letters[i]] += " " + ln.trim(); break; }

        }

      } else {

        stemLines.push(ln);

      }

    }

    let stem = stemLines.join("\n").trim();

    const choiceCount = Object.values(choices).filter(Boolean).length;

    if(!stem && choiceCount===0) continue;

    const issues = [];

    if(!stem){

      stem = "[ไม่พบข้อความโจทย์ — กรุณาตรวจ PDF ต้นฉบับ]";

      issues.push("ไม่พบข้อความโจทย์");

    }

    if(choiceCount===0) issues.push("ไม่พบตัวเลือก");

    else if(choiceCount<2) issues.push(`พบตัวเลือกเพียง ${choiceCount} ตัว`);

    questions.push({

      id: uid(), number: b.number, sourceNumber:b.number, stem, choices, issues,

      correct: null, explanation: "", image: null

    });

  }

  const usedNumbers = new Set();

  questions.forEach(q=>{

    if(usedNumbers.has(q.number)) q.issues.push(`เลขข้อ ${q.number} ซ้ำในข้อความต้นฉบับ`);

    usedNumbers.add(q.number);

  });

  return questions;

}


/* ---------------- flexible answer-key parser ---------------- */

let pendingKeyImport = null;


function parseAnswerKeyFlexible(text){

  const lines = text.split("\n");

  const blocks = [];

  let current = null;

  const numRe = /^\s*#{0,4}\s*\**\s*(?:ข้อ(?:ที่)?\s*)?(\d{1,3})\s*[\.\):\：]?\s*\**\s*(.*)$/;

  for(const raw of lines){

    const m = raw.match(numRe);

    const n = m ? parseInt(m[1],10) : NaN;

    if(m && n>0 && n<300){

      if(current) blocks.push(current);

      current = {number:m[1], lines:[m[2]]};

    } else if(current){

      current.lines.push(raw);

    }

  }

  if(current) blocks.push(current);


  const explicitAnsRe = /(?:ตอบ|เฉลย|answer|ans|correct\s*answer)\s*[:\-]?\s*\**\(?([A-E])\)?\**/i;

  const bareLetterRe = /^\**\(?([A-E])\)?\**\s*[:\.\-]?\s*$/;

  const leadLetterRe = /^\**\(?([A-E])\)?\**[\s\.\:\-]/;

  const explPrefixRe = /^\s*\**(เหตุผล|คำอธิบาย|explanation|rationale)\s*[:\-]\s*\**/i;


  return blocks.map(b=>{

    const full = b.lines.slice();

    let letter = null, ansLineIndex = -1;


    for(let i=0;i<full.length;i++){

      const m = full[i].match(explicitAnsRe);

      if(m){ letter = m[1].toUpperCase(); ansLineIndex = i; break; }

    }

    if(!letter){

      for(let i=0;i<full.length;i++){

        const trimmed = full[i].trim();

        if(!trimmed) continue;

        const m = trimmed.match(bareLetterRe) || trimmed.match(leadLetterRe);

        if(m){ letter = m[1].toUpperCase(); ansLineIndex = i; }

        break;

      }

    }


    let explLines = full.slice();

    if(ansLineIndex>=0){

      let ln = explLines[ansLineIndex];

      ln = ln.replace(explicitAnsRe,"").replace(bareLetterRe,"").replace(leadLetterRe,"").trim();

      explLines[ansLineIndex] = ln;

    }

    let explanation = explLines.join("\n").trim();

    explanation = explanation.replace(explPrefixRe,"").trim();

    explanation = explanation.replace(/\n{3,}/g,"\n\n");


    return {number: b.number, letter, explanation};

  });

}

async function exportJSON(data, filename){

  const json = JSON.stringify(data, null, 2);

  if(window.showSaveFilePicker){

    try{

      const handle = await window.showSaveFilePicker({

        suggestedName: filename,

        types: [{description:"JSON", accept:{"application/json":[".json"]}}]

      });

      const writable = await handle.createWritable();

      await writable.write(json);

      await writable.close();

      return;

    }catch(e){

      if(e && e.name === "AbortError") return; 

    }

  }

  try{

    const blob = new Blob([json], {type:"application/json"});

    const file = new File([blob], filename, {type:"application/json"});

    if(navigator.canShare && navigator.canShare({files:[file]})){

      await navigator.share({files:[file], title:filename});

      return;

    }

  }catch(e){

    if(e && e.name === "AbortError") return;

  }

  const blob = new Blob([json], {type:"application/json"});

  const a = document.createElement("a");

  a.href = URL.createObjectURL(blob);

  a.download = filename;

  a.click();

}

function compressImageFile(file, maxW){

  return new Promise((resolve, reject)=>{

    const reader = new FileReader();

    reader.onerror = reject;

    reader.onload = ()=>{

      const img = new Image();

      img.onerror = reject;

      img.onload = ()=>{

        const scale = Math.min(1, (maxW||1000)/img.width);

        const w = Math.round(img.width*scale);

        const h = Math.round(img.height*scale);

        const canvas = document.createElement("canvas");

        canvas.width = w; canvas.height = h;

        canvas.getContext("2d").drawImage(img,0,0,w,h);

        resolve(canvas.toDataURL("image/jpeg", 0.72));

      };

      img.src = reader.result;

    };

    reader.readAsDataURL(file);

  });

}

function shuffleArray(arr){

  const a = arr.slice();

  for(let i=a.length-1;i>0;i--){

    const j = Math.floor(Math.random()*(i+1));

    [a[i],a[j]] = [a[j],a[i]];

  }

  return a;

}

  const found = findExam(examId);

    const fa = state.quiz.mode === "practice" ? state.quiz.firstAttempt[id] : null;

    const userAns = state.quiz.mode === "practice" ? (fa ? fa.answer : null) : (state.quiz.answers[q.id] || null);

    const isCorrect = state.quiz.mode === "practice" ? !!(fa && fa.correct===true) : isAnswerCorrect(q, userAns);

    return {q, userAns, isCorrect};

  });

  const keyed = rows.filter(r=>r.q.correct).length;

  const correctCount = rows.filter(r=>r.isCorrect).length;


  if(!state.quiz.isReview){

    exam.history = exam.history || {exam:[], practice:[]};

    exam.history[state.quiz.mode] = exam.history[state.quiz.mode] || [];

    exam.history[state.quiz.mode].unshift({date: Date.now(), correctCount, keyed, total: rows.length});

    exam.history[state.quiz.mode] = exam.history[state.quiz.mode].slice(0,20);

    exam.lastFull = exam.lastFull || {};

    exam.lastFull[state.quiz.mode] = { date: Date.now(), rows: rows.map(r=>({qId: r.q.id, userAns: r.userAns})) };

    save();

  }


  state.lastResult = {

    examId: exam.id, examName: exam.name, mode: state.quiz.mode, isReview: !!state.quiz.isReview,

    flaggedIds: (state.quiz.flagged || []).slice(), total: rows.length, keyed, correctCount, rows

  };

  state.activeTab = "result"; render();

}
